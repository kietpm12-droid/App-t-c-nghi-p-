    <div class="description">
      Nhập số điện thoại để tra cứu dữ liệu
    </div>

    <input
      type="tel"
      id="phone"
      placeholder="Nhập số điện thoại..."
    >

    <button onclick="searchCustomer()">
      Tra cứu
    </button>

    <div id="result"></div>

  </div>

  <script>

    // SAU KHI DEPLOY BACKEND, THAY URL NÀY
    const API_URL = "https://YOUR-BACKEND-URL.onrender.com";

    async function searchCustomer() {

      const phone = document.getElementById("phone").value.trim();
      const result = document.getElementById("result");

      if (!phone) {
        result.innerHTML =
          '<div class="error">Vui lòng nhập số điện thoại.</div>';
        return;
      }

      result.innerHTML =
        '<div class="loading">Đang tra cứu...</div>';

      try {

        const response = await fetch(
          `${API_URL}/api/search?phone=${encodeURIComponent(phone)}`
        );

        const data = await response.json();

        if (!data.success) {
          result.innerHTML =
            `<div class="error">${data.message}</div>`;
          return;
        }

        if (!data.found) {
          result.innerHTML =
            `<div class="error">${data.message}</div>`;
          return;
        }

        result.innerHTML = `
          <div class="card">
            <strong>Họ tên:</strong> ${data.data.name}<br>
            <strong>Số điện thoại:</strong> ${data.data.phone}<br>
            <strong>Ghi chú:</strong> ${data.data.note}
          </div>
        `;

      } catch (error) {

        console.error(error);

        result.innerHTML = `
          <div class="error">
            Không kết nối được với máy chủ.
          </div>
        `;

      }
    }

  </script>
