  <script>
    let base64Image = "";

    // Toggle Sidebar
    const sidebar = document.getElementById('sidebar');
    const overlay = document.getElementById('overlay');
    document.getElementById('openSidebar').addEventListener('click', () => {
      sidebar.classList.add('active');
      overlay.classList.add('active');
    });
    document.getElementById('closeSidebar').addEventListener('click', closeMenu);
    overlay.addEventListener('click', closeMenu);

    function closeMenu() {
      sidebar.classList.remove('active');
      overlay.classList.remove('active');
    }

    // Xử lý nén & Xem trước Ảnh Nhà
    document.getElementById('custHouseImg').addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(event) {
          const img = new Image();
          img.onload = function() {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            const maxWidth = 600;
            const scaleSize = maxWidth / img.width;
            canvas.width = maxWidth;
            canvas.height = img.height * scaleSize;
            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
            
            base64Image = canvas.toDataURL('image/jpeg', 0.7);
            const preview = document.getElementById('previewImg');
            preview.src = base64Image;
            preview.style.display = 'block';
          }
          img.src = event.target.result;
        }
        reader.readAsDataURL(file);
      }
    });

    // Định vị GPS
    document.getElementById('btnGps').addEventListener('click', function() {
      if (!navigator.geolocation) {
        alert("Trình duyệt không hỗ trợ GPS.");
        return;
      }
      alert("Đang xác định vị trí GPS...");
      navigator.geolocation.getCurrentPosition(
        (pos) => {
          const lat = pos.coords.latitude;
          const lng = pos.coords.longitude;
          document.getElementById('geoCoords').value = `${lat},${lng}`;
          alert(`Đã lấy vị trí: ${lat.toFixed(5)}, ${lng.toFixed(5)}`);
        },
        (err) => {
          alert("Không thể lấy vị trí. Hãy đảm bảo bạn đã bật GPS và cho phép truy cập quyền vị trí.");
        },
        { enableHighAccuracy: true, timeout: 10000 }
      );
    });

    // Hiển thị danh sách & Tổng số KH
    function renderCustomers() {
      const listDiv = document.getElementById('customerList');
      const list = JSON.parse(localStorage.getItem('my_customers_v2') || '[]');

      document.getElementById('totalCount').innerText = `📌 Tổng số KH đã tác nghiệp: ${list.length}`;

      if (list.length === 0) {
        listDiv.innerHTML = '<p style="text-align: center; color: #777;">Chưa có dữ liệu tác nghiệp nào.</p>';
        return;
      }

      listDiv.innerHTML = list.map((item, index) => {
        let mapEmbed = '';
        if (item.coords) {
          mapEmbed = `<iframe class="map-frame" src="https://maps.google.com/maps?q=${item.coords}&z=15&output=embed"></iframe>`;
        }

        let imgTag = '';
        if (item.image) {
          imgTag = `<p><strong>Ảnh nhà:</strong></p><img src="${item.image}" class="house-img-list" alt="Ảnh nhà">`;
        }

        return `
          <div class="customer-item">
            <button class="btn-del" onclick="deleteCustomer(${index})">Xóa</button>
            <strong>${index + 1}. ${item.name}</strong> (CCCD: ***${item.cccd})
            <br><span>SĐT: ${item.phone}</span>
            <br><span class="tag-status">${item.status}</span>
            <p><strong>Địa chỉ nhà:</strong> ${item.address || 'Chưa nhập'}</p>
            <p><strong>Ghi chú:</strong> ${item.note || 'Không có'}</p>
            ${imgTag}
            ${mapEmbed}
            <p style="font-size: 11px; color: #888; margin-top: 6px;">⏱️ ${item.time}</p>
          </div>
        `;
      }).join('');
    }

    // Lưu thông tin
    document.getElementById('workForm').addEventListener('submit', function(e) {
      e.preventDefault();

      const newCustomer = {
        name: document.getElementById('custName').value.trim(),
        cccd: document.getElementById('custCccd').value.trim(),
        phone: document.getElementById('custPhone').value.trim(),
        address: document.getElementById('custAddress').value.trim(),
        coords: document.getElementById('geoCoords').value,
        status: document.getElementById('custStatus').value,
        image: base64Image,
        note: document.getElementById('custNote').value.trim(),
        time: new Date().toLocaleString('vi-VN')
      };

      const currentList = JSON.parse(localStorage.getItem('my_customers_v2') || '[]');
      currentList.unshift(newCustomer);
      localStorage.setItem('my_customers_v2', JSON.stringify(currentList));

      // Reset Form
      this.reset();
      base64Image = "";
      document.getElementById('previewImg').style.display = 'none';
      document.getElementById('geoCoords').value = '';

      alert("Đã lưu thành công!");
      renderCustomers();
    });

    // Xóa từng khách hàng
    function deleteCustomer(index) {
      if (confirm("Bạn có chắc muốn xóa khách hàng này?")) {
        const currentList = JSON.parse(localStorage.getItem('my_customers_v2') || '[]');
        currentList.splice(index, 1);
        localStorage.setItem('my_customers_v2', JSON.stringify(currentList));
        renderCustomers();
      }
    }

    // Xuất Excel / CSV
    document.getElementById('btnExport').addEventListener('click', function() {
      const list = JSON.parse(localStorage.getItem('my_customers_v2') || '[]');
      if (list.length === 0) {
        alert("Chưa có dữ liệu để xuất file!");
        return;
      }

      let csvContent = "\uFEFFTên KH,4 Số Cuối CCCD,Số Điện Thoại,Tình Trạng,Địa Chỉ Nhà,Tọa Độ GPS,Ghi Chú,Thời Gian\n";
      list.forEach(item => {
        const row = [
          `"${item.name}"`,
          `"${item.cccd}"`,
          `"${item.phone}"`,
          `"${item.status}"`,
          `"${item.address}"`,
          `"${item.coords}"`,
          `"${item.note}"`,
          `"${item.time}"`
        ];
        csvContent += row.join(",") + "\n";
      });

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.setAttribute("href", url);
      link.setAttribute("download", `Tac_nghiep_${new Date().toISOString().slice(0,10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    });

    window.onload = renderCustomers;
  </script>
