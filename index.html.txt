    // For Firebase JS SDK v7.20.0 and later, measurementId is optional

// Xử lý Đăng nhập dạng Chuyển hướng (Tối ưu cho cả Điện thoại & Máy tính)

    // Xử lý nén ảnh
    document.getElementById('custHouseImg').addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(event) {
          const img = new Image();
          img.onload = function() {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            const maxWidth = 500;
            const scaleSize = maxWidth / img.width;
            canvas.width = maxWidth;
            canvas.height = img.height * scaleSize;
            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
            
            base64Image = canvas.toDataURL('image/jpeg', 0.6);
            const preview = document.getElementById('previewImg');
            preview.src = base64Image;
            preview.style.display = 'block';
          }
          img.src = event.target.result;
        }
        reader.readAsDataURL(file);
      }
    });

    // Định vị GPS
    document.getElementById('btnGps').addEventListener('click', function() {
      const statusDiv = document.getElementById('gpsStatusText');
      if (!navigator.geolocation) {
        statusDiv.style.color = 'red';
        statusDiv.innerText = "❌ Trình duyệt không hỗ trợ GPS.";
        return;
      }
      statusDiv.style.color = '#2980b9';
      statusDiv.innerText = "⏳ Đang kết nối định vị GPS...";
      navigator.geolocation.getCurrentPosition(
        (pos) => {
          const lat = pos.coords.latitude;
          const lng = pos.coords.longitude;
          document.getElementById('geoCoords').value = `${lat},${lng}`;
          statusDiv.style.color = '#27ae60';
          statusDiv.innerText = `✅ Đã lấy vị trí: ${lat.toFixed(5)}, ${lng.toFixed(5)}`;
        },
        (err) => {
          statusDiv.style.color = 'red';
          statusDiv.innerText = "❌ Lỗi không lấy được vị trí GPS.";
        },
        { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
      );
    });

    // Tải dữ liệu Đám Mây Mãi Mãi (Realtime cho cả Team)
    function loadCustomersFromCloud() {
      db.collection("customers").orderBy("createdAt", "desc")
        .onSnapshot(snapshot => {
          globalCustomers = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
          renderCustomers(globalCustomers);
        });
    }

    // Hiển thị danh sách Gọn (Nhấn vào mới hiện chi tiết)
    function renderCustomers(list) {
      const listDiv = document.getElementById('customerList');
      document.getElementById('totalCount').innerText = `📌 Tổng số KH: ${list.length}`;

      if (list.length === 0) {
        listDiv.innerHTML = '<p style="text-align: center; color: #777;">Chưa có dữ liệu tác nghiệp.</p>';
        return;
      }

      listDiv.innerHTML = list.map((item, index) => {
        let mapEmbed = item.coords ? `<iframe class="map-frame" src="https://maps.google.com/maps?q=${item.coords}&z=15&output=embed"></iframe>` : '';
        let imgTag = item.image ? `<p><strong>Ảnh nhà:</strong></p><img src="${item.image}" class="house-img-list">` : '';

        return `
          <div class="customer-item">
            <div class="customer-header" onclick="toggleDetail('detail-${index}')">
              <div>
                <div class="cust-title">${item.name} (***${item.cccd})</div>
                <div class="cust-sub">📞 ${item.phone} | 📍 ${item.status}</div>
              </div>
              <span style="font-size: 12px; color: #888;">▼</span>
            </div>
            
            <div class="customer-details" id="detail-${index}">
              <div style="text-align: right; margin-bottom: 6px;">
                <button class="btn-del" onclick="deleteCustomer('${item.id}')">Xóa dữ liệu</button>
              </div>
              <span class="tag-status">${item.status}</span>
              <p><strong>Địa chỉ:</strong> ${item.address || 'Chưa nhập'}</p>
              <p><strong>Ghi chú:</strong> ${item.note || 'Không có'}</p>
              <p><strong>Người nhập:</strong> ${item.createdBy || 'N/A'}</p>
              ${imgTag}
              ${mapEmbed}
              <p style="font-size: 11px; color: #888; margin-top: 6px;">⏱️ ${item.time}</p>
            </div>
          </div>
        `;
      }).join('');
    }

    // Ẩn/Hiện chi tiết khách hàng
    function toggleDetail(id) {
      const detailDiv = document.getElementById(id);
      detailDiv.classList.toggle('show');
    }

    // Tìm kiếm khách hàng theo tên hoặc SĐT
    function filterCustomers() {
      const keyword = document.getElementById('searchInput').value.toLowerCase().trim();
      const filtered = globalCustomers.filter(item => 
        item.name.toLowerCase().includes(keyword) || item.phone.includes(keyword)
      );
      renderCustomers(filtered);
    }

    // Lưu dữ liệu vĩnh viễn lên Đám Mây (Cloud)
    document.getElementById('workForm').addEventListener('submit', function(e) {
      e.preventDefault();

      const newCustomer = {
        name: document.getElementById('custName').value.trim(),
        cccd: document.getElementById('custCccd').value.trim(),
        phone: document.getElementById('custPhone').value.trim(),
        address: document.getElementById('custAddress').value.trim(),
        coords: document.getElementById('geoCoords').value,
        status: document.getElementById('custStatus').value,
        image: base64Image,
        note: document.getElementById('custNote').value.trim(),
        time: new Date().toLocaleString('vi-VN'),
        createdBy: auth.currentUser.email,
        createdAt: firebase.firestore.FieldValue.serverTimestamp()
      };

      db.collection("customers").add(newCustomer)
        .then(() => {
          alert("✅ Đã lưu vĩnh viễn lên hệ thống!");
          this.reset();
          base64Image = "";
          document.getElementById('previewImg').style.display = 'none';
          document.getElementById('geoCoords').value = '';
          document.getElementById('gpsStatusText').innerText = '';
        })
        .catch(err => alert("Lỗi lưu dữ liệu: " + err.message));
    });

    // Xóa khách hàng trên Cloud
    function deleteCustomer(id) {
      if (confirm("Bạn có chắc muốn xóa dữ liệu này khỏi hệ thống chung?")) {
        db.collection("customers").doc(id).delete();
      }
    }

    // Xuất dữ liệu Excel / CSV
    document.getElementById('btnExport').addEventListener('click', function() {
      if (globalCustomers.length === 0) {
        alert("Chưa có dữ liệu để xuất file!");
        return;
      }
      let csvContent = "\uFEFFTên KH,4 Số Cuối CCCD,Số Điện Thoại,Tình Trạng,Địa Chỉ Nhà,Tọa Độ GPS,Ghi Chú,Người Nhập,Thời Gian\n";
      globalCustomers.forEach(item => {
        const row = [
          `"${item.name}"`,
          `"${item.cccd}"`,
          `"${item.phone}"`,
          `"${item.status}"`,
          `"${item.address}"`,
          `"${item.coords}"`,
          `"${item.note}"`,
          `"${item.createdBy}"`,
          `"${item.time}"`
        ];
        csvContent += row.join(",") + "\n";
      });

      const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.setAttribute("href", url);
      link.setAttribute("download", `Tac_nghiep_${new Date().toISOString().slice(0,10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    });
