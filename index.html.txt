  <script>
    let customers = JSON.parse(localStorage.getItem('field_customers') || '[]');

    // Toggle menu 3 gạch
    function toggleMenu() {
      const sidebar = document.getElementById('sidebar');
      const overlay = document.getElementById('overlay');
      sidebar.classList.toggle('open');
      overlay.classList.toggle('active');
    }

    function scrollToSection(id) {
      toggleMenu();
      document.getElementById(id).scrollIntoView({ behavior: 'smooth' });
    }

    // Lấy tọa độ GPS từ trình duyệt/điện thoại
    function getGPS() {
      const gpsInput = document.getElementById('gps');
      if (navigator.geolocation) {
        gpsInput.placeholder = "Đang lấy vị trí...";
        navigator.geolocation.getCurrentPosition(
          (position) => {
            const lat = position.coords.latitude.toFixed(6);
            const lng = position.coords.longitude.toFixed(6);
            gpsInput.value = `${lat}, ${lng}`;
          },
          (error) => {
            alert("Không thể lấy tọa độ. Vui lòng bật vị trí GPS trên thiết bị.");
            gpsInput.placeholder = "Lỗi lấy GPS";
          },
          { enableHighAccuracy: true, timeout: 10000 }
        );
      } else {
        alert("Trình duyệt không hỗ trợ định vị GPS.");
      }
    }

    // Xử lý gửi form
    document.getElementById('debt-form').addEventListener('submit', function(e) {
      e.preventDefault();
      
      const fileInput = document.getElementById('image');
      const file = fileInput.files[0];

      if (file) {
        const reader = new FileReader();
        reader.onloadend = function() {
          saveCustomerData(reader.result);
        };
        reader.readAsDataURL(file);
      } else {
        saveCustomerData('');
      }
    });

    function saveCustomerData(imageData) {
      const newCustomer = {
        id: Date.now(),
        name: document.getElementById('name').value,
        address: document.getElementById('address').value,
        gps: document.getElementById('gps').value,
        status: document.getElementById('status').value,
        notes: document.getElementById('notes').value,
        image: imageData,
        date: new Date().toLocaleString('vi-VN')
      };

      customers.unshift(newCustomer);
      localStorage.setItem('field_customers', JSON.stringify(customers));
      
      document.getElementById('debt-form').reset();
      renderCustomers();
      alert("Đã lưu thông tin tác nghiệp thành công!");
    }

    // Trích xuất danh sách và bộ lọc tìm kiếm
    function renderCustomers() {
      const listContainer = document.getElementById('customer-list');
      const searchKeyword = document.getElementById('search-input').value.toLowerCase();
      
      const filtered = customers.filter(c => c.name.toLowerCase().includes(searchKeyword));
      
      document.getElementById('total-count').innerText = customers.length;
      listContainer.innerHTML = '';

      if (filtered.length === 0) {
        listContainer.innerHTML = '<p style="text-align:center; color:#94a3b8; padding:10px;">Chưa có dữ liệu phù hợp.</p>';
        return;
      }

      filtered.forEach(item => {
        const div = document.createElement('div');
        div.className = 'customer-item';
        
        let badgeClass = 'badge-pending';
        if (item.status === 'Đã thu tiền') badgeClass = 'badge-paid';
        if (item.status === 'Không gặp/Vắng nhà') badgeClass = 'badge-unreachable';

        const gpsLink = item.gps ? `<a href="https://maps.google.com/?q=${item.gps}" target="_blank" style="color:#0284c7; text-decoration:none;">📍 ${item.gps}</a>` : 'Chưa lấy vị trí';

        div.innerHTML = `
          <img src="${item.image || 'https://via.placeholder.com/60?text=No+Img'}" alt="Photo">
          <div class="customer-info">
            <h4>${item.name}</h4>
            <p>🏠 ${item.address || 'Chưa cập nhật'}</p>
            <p>${gpsLink}</p>
            <p>📝 ${item.notes || 'Không có ghi chú'}</p>
            <span class="badge ${badgeClass}">${item.status}</span>
            <span style="font-size: 10px; color: #94a3b8; display: block; margin-top: 2px;">${item.date}</span>
          </div>
        `;
        listContainer.appendChild(div);
      });
    }

    // Xuất file CSV/Excel
    function exportCSV() {
      if (customers.length === 0) {
        alert("Chưa có dữ liệu để xuất!");
        return;
      }
      
      let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
      csvContent += "Họ tên,Địa chỉ,GPS,Tình trạng,Ghi chú,Ngày cập nhật\n";

      customers.forEach(c => {
        const row = [
          `"${c.name}"`,
          `"${c.address}"`,
          `"${c.gps}"`,
          `"${c.status}"`,
          `"${c.notes}"`,
          `"${c.date}"`
        ].join(",");
        csvContent += row + "\n";
      });

      const encodedUri = encodeURI(csvContent);
      const link = document.createElement("a");
      link.setAttribute("href", encodedUri);
      link.setAttribute("download", `Danh_sach_KH_${new Date().toISOString().slice(0,10)}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    // Khởi tạo hiển thị ban đầu
    renderCustomers();
  </script>
